> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make Klyntbot invisible on M1 Air 8GB — <160MB idle RAM, <2% background CPU, <1.8s cold launch — through 3-tier feature architecture, runtime pressure awareness, compile-time gating, and frontend code splitting.

**Architecture:** Approach B (Measure-Cut-Measure) + selective C (compile-time gating). Three tiers: always-loaded core (chat, tasks, basic memory, coaching signals, basic finance), lazy-init services (LanceDB, voice, launcher cache, embeddings), and compile-time gated power features (WASM, email, autotuner, mirror narratives). A SystemPressureMonitor reads macOS native signals and throttles all background work dynamically.

**Tech Stack:** Rust (tokio, sqlx, lancedb, tauri), macOS native APIs (objc2, dispatch_source), React 19 (Vite, React.lazy), Biome 2.0

```rust
use std::sync::atomic::{AtomicU8, Ordering};
use std::sync::Arc;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
#[repr(u8)]
pub enum PressureLevel {
    Normal = 0,
    Elevated = 1,
    Critical = 2,
}

impl PressureLevel {
    fn from_u8(v: u8) -> Self {
        match v {
            1 => Self::Elevated,
            2 => Self::Critical,
            _ => Self::Normal,
        }
    }
}

/// Computes aggregate pressure from individual signals.
/// Memory pressure: 0=normal, 1=warn, 2=critical (macOS dispatch_source values).
/// CPU: percentage (0-100).
/// Thermal: 0=nominal, 1=fair, 2=serious, 3=critical (NSProcessInfo values).
pub fn compute_pressure(memory_pressure: u8, cpu_percent: u8, thermal_state: u8) -> PressureLevel {
    // Critical: any signal is critical
    if memory_pressure >= 2 || cpu_percent > 70 || thermal_state >= 3 {
        return PressureLevel::Critical;
    }
    // Elevated: any signal is warning-level
    if memory_pressure >= 1 || cpu_percent >= 45 || thermal_state >= 2 {
        return PressureLevel::Elevated;
    }
    PressureLevel::Normal
}

        assert_eq!(compute_pressure(0, 20, 0), PressureLevel::Normal);

        assert_eq!(compute_pressure(1, 20, 0), PressureLevel::Elevated);
```

```rust
pub struct SystemPressureMonitor {
    level: Arc<AtomicU8>,
    shutdown: tokio::sync::watch::Sender<bool>,
}

impl SystemPressureMonitor {
    /// Starts monitoring. Polls system stats every `poll_interval`.
    /// Returns immediately; polling happens in a background tokio task.
    pub fn start(poll_interval: std::time::Duration) -> Self {
        let level = Arc::new(AtomicU8::new(PressureLevel::Normal as u8));
        let (shutdown_tx, mut shutdown_rx) = tokio::sync::watch::channel(false);
        let level_clone = Arc::clone(&level);

        tokio::spawn(async move {
            loop {
                tokio::select! {
                    _ = shutdown_rx.changed() => break,
                    _ = tokio::time::sleep(poll_interval) => {
                        let (mem, cpu, thermal) = read_system_stats();
                        let p = compute_pressure(mem, cpu, thermal);
                        level_clone.store(p as u8, Ordering::Relaxed);
                    }
                }
            }
        });

        Self { level, shutdown: shutdown_tx }
    }

    pub fn current_pressure(&self) -> PressureLevel {
        PressureLevel::from_u8(self.level.load(Ordering::Relaxed))
    }

    pub fn is_under_pressure(&self) -> bool {
        self.current_pressure() != PressureLevel::Normal
    }

    pub fn should_defer_background_work(&self) -> bool {
        matches!(self.current_pressure(), PressureLevel::Elevated | PressureLevel::Critical)
    }
}

impl Drop for SystemPressureMonitor {
    fn drop(&mut self) {
        let _ = self.shutdown.send(true);
    }
}

/// Reads macOS system stats. Returns (memory_pressure, cpu_percent, thermal_state).
fn read_system_stats() -> (u8, u8, u8) {
    let mem = read_memory_pressure();
    let cpu = read_cpu_load();
    let thermal = read_thermal_state();
    (mem, cpu, thermal)
}
```

```rust
/// Returns macOS memory pressure level: 0=normal, 1=warn, 2=critical.
/// Uses kern.memorystatus_vm_pressure_level sysctl.
fn read_memory_pressure() -> u8 {
    use std::mem::MaybeUninit;

    // kern.memorystatus_vm_pressure_level
    let mut mib = [libc::CTL_KERN, libc::KERN_MEMORYSTATUS_VM_PRESSURE_LEVEL];
    let mut pressure_level: i32 = 0;
    let mut size = std::mem::size_of::<i32>();

    let ret = unsafe {
        libc::sysctl(
            mib.as_mut_ptr(),
            mib.len() as u32,
            &mut pressure_level as *mut _ as *mut libc::c_void,
            &mut size,
            std::ptr::null_mut(),
            0,
        )
    };

    if ret != 0 {
        tracing::warn!("sysctl memorystatus_vm_pressure_level failed");
        return 0; // assume normal on failure
    }

    // macOS: 1=normal, 2=warn, 4=critical
    match pressure_level {
        4 => 2, // critical
        2 => 1, // warn
        _ => 0, // normal
    }
}
```

Note: If `KERN_MEMORYSTATUS_VM_PRESSURE_LEVEL` is not available as a Rust constant, use the raw sysctl name `"kern.memorystatus_vm_pressure_level"` via `sysctlbyname`. Check the platform-macos crate's existing libc usage for patterns.

```rust
/// Returns 1-minute CPU load average as a percentage (0-100).
fn read_cpu_load() -> u8 {
    let mut load: [libc::c_double; 3] = [0.0; 3];
    let ret = unsafe { libc::getloadavg(load.as_mut_ptr(), 1) };
    if ret < 1 {
        return 0;
    }
    // Load average is per-core. Normalize to percentage of available cores.
    let ncpu = num_cpus() as f64;
    let pct = ((load[0] / ncpu) * 100.0).clamp(0.0, 100.0) as u8;
    pct
}

fn num_cpus() -> usize {
    std::thread::available_parallelism()
        .map(|n| n.get())
        .unwrap_or(1)
}
```

```rust
use platform_macos::pressure::{PressureLevel, SystemPressureMonitor};
use std::sync::Arc;
use tokio::sync::Semaphore;

/// Pressure-aware concurrency limiter for background LLM calls.
/// Fixed at 2 permits. Pressure gates access:
/// - Normal: acquire (wait for permit, max 2 concurrent)
/// - Elevated: try_acquire (at most 1 — don't wait if busy)
/// - Critical: defer entirely
pub struct BackgroundSemaphore {
    semaphore: Arc<Semaphore>,
    pressure: Arc<SystemPressureMonitor>,
}

pub enum AcquireResult {
    Acquired(tokio::sync::OwnedSemaphorePermit),
    Deferred,
}

impl BackgroundSemaphore {
    pub fn new(pressure: Arc<SystemPressureMonitor>) -> Self {
        Self {
            semaphore: Arc::new(Semaphore::new(2)),
            pressure,
        }
    }

    /// Attempts to acquire a permit based on current system pressure.
    /// Returns Deferred if the job should be rescheduled.
    pub async fn acquire(&self) -> AcquireResult {
        match self.pressure.current_pressure() {
            PressureLevel::Critical => AcquireResult::Deferred,
            PressureLevel::Elevated => {
                match self.semaphore.clone().try_acquire_owned() {
                    Ok(permit) => AcquireResult::Acquired(permit),
                    Err(_) => AcquireResult::Deferred,
                }
            }
            PressureLevel::Normal => {
                match self.semaphore.clone().acquire_owned().await {
                    Ok(permit) => AcquireResult::Acquired(permit),
                    Err(_) => AcquireResult::Deferred,
                }
            }
        }
    }
}
```

```rust
use std::sync::Arc;
use std::time::{Duration, Instant};
use tokio::sync::RwLock;
use platform_macos::pressure::{PressureLevel, SystemPressureMonitor};

/// Tracks last access time for idle-release decisions.
pub struct IdleTracker {
    last_access: Arc<RwLock<Instant>>,
}

impl IdleTracker {
    pub fn new() -> Self {
        Self {
            last_access: Arc::new(RwLock::new(Instant::now())),
        }
    }

    pub async fn touch(&self) {
        *self.last_access.write().await = Instant::now();
    }

    pub async fn idle_duration(&self) -> Duration {
        self.last_access.read().await.elapsed()
    }
}

/// Spawns an idle-release monitor for a lazy service.
/// When idle exceeds `normal_threshold`, sets the Option<T> to None.
/// Under Elevated pressure, threshold shortens to `elevated_threshold`.
/// Under Critical, immediate release.
pub fn spawn_idle_release<T: Send + Sync + 'static>(
    service: Arc<RwLock<Option<T>>>,
    tracker: Arc<IdleTracker>,
    pressure: Arc<SystemPressureMonitor>,
    normal_threshold: Duration,
    elevated_threshold: Duration,
    label: &'static str,
) -> tokio::task::JoinHandle<()> {
    tokio::spawn(async move {
        loop {
            tokio::time::sleep(Duration::from_secs(60)).await;

            let threshold = match pressure.current_pressure() {
                PressureLevel::Critical => Duration::ZERO,
                PressureLevel::Elevated => elevated_threshold,
                PressureLevel::Normal => normal_threshold,
            };

            if tracker.idle_duration().await >= threshold {
                let mut guard = service.write().await;
                if guard.is_some() {
                    *guard = None;
                    tracing::info!("Lazy service '{label}' released after idle");
                }
            }
        }
    })
}
```
